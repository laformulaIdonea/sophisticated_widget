        let audioCtx = null;


        function playChimeSequence() {

            try {

                if (!audioCtx) {

                    audioCtx = new (window.AudioContext || window.webkitAudioContext)();

                }

                if (audioCtx.state === 'suspended') {

                    audioCtx.resume();

                }


                const repetitions = 3;

                const repetitionDelay = 1.2;


                for (let r = 0; r < repetitions; r++) {

                    const loopOffset = r * repetitionDelay;

                    const now = audioCtx.currentTime + loopOffset;

                    const frequencies = [659.25, 783.99, 880.00, 1046.50];

                    const noteDuration = 0.22;


                    frequencies.forEach((freq, index) => {

                        const osc = audioCtx.createOscillator();

                        const gain = audioCtx.createGain();


                        osc.type = 'sine';

                        osc.frequency.setValueAtTime(freq, now + (index * noteDuration));


                        const startTime = now + (index * noteDuration);

                        gain.gain.setValueAtTime(0, startTime);

                        gain.gain.linearRampToValueAtTime(0.18, startTime + 0.02);

                        gain.gain.exponentialRampToValueAtTime(0.001, startTime + noteDuration * 1.8);


                        osc.connect(gain);

                        gain.connect(audioCtx.destination);


                        osc.start(startTime);

                        osc.stop(startTime + noteDuration * 2);

                    });

                }

            } catch (e) {

                console.log("Audio playback prevented or unsupported", e);

            }

        }


        let currentMode = 0;

        const totalModes = 3;

        const modeTitles = ["Clock & Alarm", "Countdown Timer", "Analog Stopwatch"];

        

        const modeViews = document.querySelectorAll('.mode-view');

        const modeTitleEl = document.getElementById('mode-title');

        const indicators = document.querySelectorAll('.indicator');


        function setMode(index) {

            currentMode = (index + totalModes) % totalModes;

            modeViews.forEach((view, idx) => {

                if (idx === currentMode) {

                    view.classList.remove('hidden');

                } else {

                    view.classList.add('hidden');

                }

            });

            modeTitleEl.textContent = modeTitles[currentMode];

            

            indicators.forEach((ind, idx) => {

                if (idx === currentMode) {

                    ind.className = "indicator w-4 h-1.5 rounded-full bg-[#3f3e3a] dark:bg-[#e0dcd0] transition-all";

                } else {

                    ind.className = "indicator w-1.5 h-1.5 rounded-full bg-[#d0cac0] dark:bg-[#3a3a3a] transition-all";

                }

            });

        }


        document.getElementById('prev-mode').addEventListener('click', () => setMode(currentMode - 1));

        document.getElementById('next-mode').addEventListener('click', () => setMode(currentMode + 1));


        let touchStartX = 0;

        const container = document.getElementById('widget-container');

        container.addEventListener('touchstart', e => { touchStartX = e.changedTouches[0].screenX; }, {passive: true});

        container.addEventListener('touchend', e => {

            let diff = e.changedTouches[0].screenX - touchStartX;

            if (Math.abs(diff) > 50) {

                if (diff > 0) setMode(currentMode - 1);

                else setMode(currentMode + 1);

            }

        }, {passive: true});


        let alarmTime = null;

        const hoursEl = document.getElementById('hours');

        const minutesEl = document.getElementById('minutes');

        const dateDisplayEl = document.getElementById('date-display');

        function updateClock() {

            const now = new Date();

            let h = now.getHours();

            const m = now.getMinutes();

            const s = now.getSeconds();

            const ampm = h >= 12 ? 'PM' : 'AM';

            

            h = h % 12;

            h = h ? h : 12;


            hoursEl.textContent = String(h).padStart(2, '0');

            minutesEl.textContent = String(m).padStart(2, '0');

            ampmEl.textContent = ampm;


            const options = { weekday: 'long', month: 'short', day: 'numeric' };

            dateDisplayEl.textContent = now.toLocaleDateString('en-US', options);


            if (alarmTime && alarmTime.active) {

                if (alarmTime.hour === h && alarmTime.minute === m && s === 0 && alarmTime.ampm === ampm) {

                    playChimeSequence();

                    alarmTime.active = false;

                    alarmTextEl.textContent = "Alarm Triggered!";

                    setTimeout(() => { alarmTextEl.textContent = `Alarm: ${String(alarmTime.hour).padStart(2,'0')}:${String(alarmTime.minute).padStart(2,'0')} ${alarmTime.ampm}`; }, 10000);

                }

            }

        }

        setInterval(updateClock, 1000);

        let chronoStartTime = 0;

        let chronoElapsedTime = 0;

        let chronoInterval = null;

        let chronoRunning = false;


        const chronoDisplay = document.getElementById('chrono-display');

        const chronoStartBtn = document.getElementById('chrono-start');

        const chronoRing = document.getElementById('chrono-ring');

        const chronoLapDisplay = document.getElementById('chrono-lap-display');


        function formatChrono(ms) {

            const totalSecs = Math.floor(ms / 1000);

            const mins = Math.floor(totalSecs / 60);

            const secs = totalSecs % 60;

            const tenths = Math.floor((ms % 1000) / 100);

            return `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}.${tenths}`;

        }


        function updateChrono() {

            const now = Date.now();

            chronoElapsedTime = now - chronoStartTime;

            chronoDisplay.textContent = formatChrono(chronoElapsedTime);


            const secsFraction = (chronoElapsedTime % 60000) / 60000;

            const dashOffset = 276.46 - (276.46 * secsFraction);

            chronoRing.style.strokeDashoffset = dashOffset;

        }


        chronoStartBtn.addEventListener('click', () => {

            if (chronoRunning) {

                clearInterval(chronoInterval);

                chronoRunning = false;

                chronoStartBtn.textContent = "Start";

            } else {

                chronoStartTime = Date.now() - chronoElapsedTime;

                chronoRunning = true;

                chronoStartBtn.textContent = "Pause";

                chronoInterval = setInterval(updateChrono, 50);

            }

        });


        document.getElementById('chrono-lap').addEventListener('click', () => {

            if (!chronoRunning && chronoElapsedTime === 0) return;

            chronoLapDisplay.textContent = `Lap: ${formatChrono(chronoElapsedTime)}`;

        });


        document.getElementById('chrono-reset').addEventListener('click', () => {

            clearInterval(chronoInterval);

            chronoRunning = false;

            chronoStartBtn.textContent = "Start";

            chronoElapsedTime = 0;

            chronoDisplay.textContent = "00:00.0";

            chronoRing.style.strokeDashoffset = 276.46;

            chronoLapDisplay.textContent = "";

        });


        const themeToggle = document.getElementById('theme-toggle');

        function applyTheme(theme) {

            const isDark = theme === 'dark';

            root.classList.toggle('seraph-dark', isDark);


            themeToggle.textContent = isDark ? '☀' : '⏾';

            themeToggle.setAttribute(

                'aria-label',

                isDark ? 'Switch to light theme' : 'Switch to dark theme'

            );

            themeToggle.title = isDark ? 'Switch to light theme' : 'Switch to dark theme';

        }


        let savedTheme = null;

        try {

            savedTheme = localStorage.getItem('seraph-theme');

        } catch (e) {}


        const systemPrefersDark =

            window.matchMedia &&

            window.matchMedia('(prefers-color-scheme: dark)').matches;


        applyTheme(savedTheme || (systemPrefersDark ? 'dark' : 'light'));


        themeToggle.addEventListener('click', () => {

            const nextTheme = root.classList.contains('seraph-dark')

                ? 'light'

                : 'dark';


            applyTheme(nextTheme);


            try {

                localStorage.setItem('seraph-theme', nextTheme);

            } catch (e) {}

        });

    </script>
